"""초기비용 4블록 순수 함수 (스펙 §4-1, design 2026-07-24). 만원 단위 정수, 구간 표기.
상수 근거·잠정치는 docs/assumptions.md. BE CostCalculator 재계산과 정합 유지."""
from __future__ import annotations

# 대표면적 = 인허가 소재지면적 영업중 중앙값 (채움률 99.8%, design 2-1)
REPRESENTATIVE_AREA_M2 = {"cafe": 29.2, "food": 55.2}
# 보증금 관행배수 (상가; 실측 보증액 0% → 관행, 전환율 병기)
DEPOSIT_MULT_LOW, DEPOSIT_MULT_HIGH = 8, 12
# 권리금: 서울 숙박·음식점 ㎡당 평균 2025 (R-ONE A_2024_00445), 업종보정·구간비
PREMIUM_PER_M2_MANWON = 72.6
PREMIUM_INDUSTRY_ADJ = {"cafe": 0.85, "food": 1.0}
PREMIUM_LOW_RATIO, PREMIUM_HIGH_RATIO = 0.72, 1.15
RENT_RATIO_CLIP = (0.5, 2.0)
# 인테리어: FTC avrgJngEtcAmt 2025 만원 (assumptions #23), ±20%
INTERIOR_MANWON = {"cafe": 2485, "food": 4595}
INTERIOR_BAND = 0.20
# 월고정비 가산형 (design 2-3)
MIN_WAGE_MONTHLY_MANWON = 210   # 최저임금 시급×209h 월환산 (2025 기준; 2026 확정 시 갱신)
STAFF_STD = {"cafe": 1.5, "food": 2.0}
UTILITY_RATE = 0.20             # 공과금·기타 = 환산임대료 비율 (KOSIS 영업비용 확정치로 교체, A0)


def converted_rent(unit_price_1000won_m2: float, industry: str) -> int:
    """환산임대료 만원/월 = 단가(천원/㎡) × 대표면적 ÷ 10."""
    return round(unit_price_1000won_m2 * REPRESENTATIVE_AREA_M2[industry] / 10)


def deposit_interval(monthly_rent: int) -> tuple[int, int]:
    return (monthly_rent * DEPOSIT_MULT_LOW, monthly_rent * DEPOSIT_MULT_HIGH)


def premium_interval(monthly_rent: int, seoul_median_rent: int, industry: str) -> tuple[int, int]:
    ratio = monthly_rent / seoul_median_rent if seoul_median_rent else 1.0
    ratio = min(max(ratio, RENT_RATIO_CLIP[0]), RENT_RATIO_CLIP[1])
    center = PREMIUM_PER_M2_MANWON * REPRESENTATIVE_AREA_M2[industry] * ratio * PREMIUM_INDUSTRY_ADJ[industry]
    return (round(center * PREMIUM_LOW_RATIO), round(center * PREMIUM_HIGH_RATIO))


def interior_interval(industry: str) -> tuple[int, int]:
    c = INTERIOR_MANWON[industry]
    return (round(c * (1 - INTERIOR_BAND)), round(c * (1 + INTERIOR_BAND)))


def monthly_fixed_cost(monthly_rent: int, industry: str) -> int:
    labor = STAFF_STD[industry] * MIN_WAGE_MONTHLY_MANWON
    utility = monthly_rent * UTILITY_RATE
    return round(monthly_rent + labor + utility)


def cost_ex_premium(deposit: tuple[int, int], interior: tuple[int, int], mfc: int) -> tuple[int, int]:
    reserve = mfc * 6
    return (deposit[0] + interior[0] + reserve, deposit[1] + interior[1] + reserve)


def cost_incl_premium(ex: tuple[int, int], premium: tuple[int, int]) -> tuple[int, int]:
    return (ex[0] + premium[0], ex[1] + premium[1])
```

"""점수화 w1~w5 — 서울 전체 백분위 (스펙 §4-3). BE ScoreLookup 정의와 동일.
교통 감쇠 500m, w1 3성분 균등평균. 종합점수는 저장 안 함(BE 파생)."""
from __future__ import annotations

import math

TRANSIT_DECAY_M = 500.0


def percentile(value: float, population: list[float]) -> float:
    """서울 전체 모집단 중 value 이하 비율 [0,1]."""
    if not population:
        return 0.0
    at_or_below = sum(1 for p in population if p <= value)
    return at_or_below / len(population)


def transit_influx(daily_riders: float, distance_m: float) -> float:
    return daily_riders * math.exp(-distance_m / TRANSIT_DECAY_M)


def demand_w1(pedestrian_pct: float, backing_pct: float, transit_pct: float) -> float:
    return (pedestrian_pct + backing_pct + transit_pct) / 3.0
```

import pandas as pd

def build_initial_cost(rent_df: pd.DataFrame, seoul_median_rent: int) -> pd.DataFrame:
    out = []
    for _, r in rent_df.iterrows():
        ind = r["industry"]
        mr = converted_rent(r["unit_price"], ind)
        dep = deposit_interval(mr); prem = premium_interval(mr, seoul_median_rent, ind)
        intr = interior_interval(ind); mfc = monthly_fixed_cost(mr, ind)
        ex = cost_ex_premium(dep, intr, mfc); incl = cost_incl_premium(ex, prem)
        out.append({"area_code": r["area_code"], "industry": ind,
            "deposit_low": dep[0], "deposit_high": dep[1],
            "premium_low": prem[0], "premium_high": prem[1],
            "interior_low": intr[0], "interior_high": intr[1],

import pandas as pd

def build_location_score(metrics: pd.DataFrame) -> pd.DataFrame:
    df = metrics.copy()
    df["influx"] = [transit_influx(r.riders, r.distance_m) for r in df.itertuples()]
    df["backing_total"] = df["backing"]
    out = []
    for ind, g in df.groupby("industry"):
        ped = g["pedestrian"].tolist(); bck = g["backing_total"].tolist()
        inf = g["influx"].tolist(); sal = g["est_sales"].tolist()
        neg_den = (-g["density"]).tolist(); gr = g["growth_rank"].tolist()
        eff = (g["est_sales"]/g["est_sales"].index.map(lambda i: 1)).tolist()  # placeholder → 실제 est_sales/rent
        for row in g.itertuples():
            w1 = demand_w1(percentile(row.pedestrian, ped),
                           percentile(row.backing_total, bck),
                           percentile(row.influx, inf))
            w2 = percentile(row.est_sales, sal)
            w3 = percentile(-row.density, neg_den)
            w4 = percentile(row.growth_rank, gr)
            w5 = percentile(row.est_sales/row.monthly_rent, [s/m for s,m in zip(g["est_sales"], g["monthly_rent"])])
            out.append({"area_code": row.area_code, "industry": ind,
                "w1": w1, "w2": w2, "w3": w3, "w4": w4, "w5": w5,
                "est_sales": int(row.est_sales), "daily_floating": int(row.daily_floating),
                "based_on_quarter": row.quarter})
    return pd.DataFrame(out)
